function Convert-HexToShortName {
    param(
        [Parameter(Mandatory)]
        [ValidatePattern('^[0-9A-Fa-f]{8}$')]
        [string]$Hex
    )

    $Base = $Alphabet.Count  # 85

    # Convert hex → 32-bit integer
    $Value = [uint32]::Parse($Hex, 'HexNumber')

    # Convert integer → base-86
    $Chars = New-Object System.Collections.Generic.List[string]
    for ($i = 0; $i -lt 5; $i++) {
        $Chars.Add($Alphabet[$Value % $Base])
        $Value = [math]::Floor($Value / $Base)
    }

    # Reverse because we built it least-significant first
    -join ($Chars.ToArray() | Sort-Object -Descending)
}

function Convert-ShortNameToHex {
    param(
        [Parameter(Mandatory)]
        [ValidateLength(5,5)]
        [string]$Short
    )

    $Base = $Alphabet.Count  # 86

    $Value = 0
    foreach ($c in $Short.ToCharArray()) {
        $Index = $Alphabet.IndexOf($c)
        if ($Index -lt 0) { throw "Invalid character '$c' in short name." }
        $Value = ($Value * $Base) + $Index
    }

    # Convert back to 8-char hex
    return ('{0:X8}' -f $Value)
}
